export const meta = {
  title: "Delete contact",
  description: "Permanently remove a contact and its associated activities.",
};

<MethodBadge method="DELETE" path="/api-contacts/:id" />

Requires the `contacts:write` scope.

<Callout type="warn" title="Destructive">
Deletion cascades to the contact's notes and activity history. Linked deals are preserved but unlink from the contact.
</Callout>

## Path parameters

<ParamTable rows={[
  { name: "id", type: "uuid", required: true, description: "The contact ID." },
]} title="Path" />

## Example request

```bash
curl -X DELETE \
  'https://api.wundertreos.com/functions/v1/api-contacts/c8b1f3a0-4d2e-4f9a-9b1c-2d4e5f6a7b8c' \
  -H 'Authorization: Bearer YOUR_ACCESS_TOKEN'
```

## Response

Returns `204 No Content` on success.
